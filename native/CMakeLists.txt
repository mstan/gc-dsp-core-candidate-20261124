get_filename_component(_gcn_dsp_root ".." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_LIST_DIR}")
set(_gcn_dsp_source "${CMAKE_CURRENT_LIST_DIR}")
include("${CMAKE_CURRENT_LIST_DIR}/adapt_hle.cmake")
set(GC_DSP_DONOR_HOST_DIR "" CACHE PATH "Pinned native DSP donor host headers")

function(gcnllr_add_native_dsp target provider catalog)
  if(TARGET "${target}")
    message(FATAL_ERROR "DSP target already exists: ${target}")
  endif()
  if(provider STREQUAL "hle")
    if(NOT TARGET bluewake_dsp_adapter OR NOT TARGET bluewake_dsp_donor OR NOT GC_DSP_DONOR_HOST_DIR)
      message(FATAL_ERROR "DSP HLE requires pinned donor targets and GC_DSP_DONOR_HOST_DIR")
    endif()
    gcnllr_adapt_dsp_rom(bluewake_dsp_donor)
    add_library("${target}" STATIC "${_gcn_dsp_source}/hle_engine.cpp")
    target_include_directories("${target}" PRIVATE "${GC_DSP_DONOR_HOST_DIR}")
    target_link_libraries("${target}" PRIVATE bluewake_dsp_adapter)
  elseif(provider STREQUAL "aot" OR provider STREQUAL "interpreter-development")
    file(GLOB _core CONFIGURE_DEPENDS "${_gcn_dsp_root}/Core/DSP/*.cpp"
      "${_gcn_dsp_root}/Core/DSP/Interpreter/*.cpp")
    add_library("${target}" STATIC ${_core} "${_gcn_dsp_root}/Core/DSP/Aot/Aot.cpp"
      "${_gcn_dsp_source}/lle_engine.cpp" "${_gcn_dsp_source}/lle_host.cpp"
      "${_gcn_dsp_root}/src/dsp_core_state.cpp")
    target_include_directories("${target}" PRIVATE "${_gcn_dsp_source}" "${_gcn_dsp_root}")
    if(CMAKE_CXX_COMPILER_FRONTEND_VARIANT STREQUAL "MSVC")
      target_compile_options("${target}" PRIVATE /FICommon/CommonTypes.h /FICommon/Logging/Log.h /FICommon/Assert.h)
    else()
      target_compile_options("${target}" PRIVATE "SHELL:-include Common/CommonTypes.h"
        "SHELL:-include Common/Logging/Log.h" "SHELL:-include Common/Assert.h")
    endif()
    if(provider STREQUAL "aot")
      if(NOT EXISTS "${catalog}/sources.cmake")
        message(FATAL_ERROR "Strict DSP AOT requires a prepared private catalog")
      endif()
      include("${catalog}/sources.cmake")
      target_sources("${target}" PRIVATE ${GCNLLR_DSP_AOT_SOURCES})
      target_compile_definitions("${target}" PRIVATE GCN_DSP_ENGINE_AOT=1
        GCNLLR_DSP_AOT=1 GCNLLR_DSP_AOT_CATALOG=1 GCNLLR_DSP_AOT_STRICT=1)
    else()
      target_compile_definitions("${target}" PRIVATE GCN_DSP_ENGINE_AOT=0)
    endif()
  else()
    message(FATAL_ERROR "Unknown DSP provider: ${provider}")
  endif()
  target_include_directories("${target}" PUBLIC "${_gcn_dsp_root}/include")
  target_compile_features("${target}" PUBLIC cxx_std_20)
endfunction()
